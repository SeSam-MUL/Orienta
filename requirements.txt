# PyTorch (GPU spherical indexing, GPU dictionary indexing, GPU forward simulation)
#
# PyTorch MUST be pinned to a +cuXXX local version, not merely offered a CUDA
# index. `--extra-index-url` makes pip consider BOTH indexes and take the
# HIGHEST version: the cu121 index stops at torch 2.5.1+cu121 (cp311/Windows)
# while PyPI is far ahead, so an unpinned `torch>=2.0` always resolved to the
# PyPI wheel -- which is CPU-only on Windows. Measured 2026-09-17 on an RTX
# 4070: `pip install -r requirements.txt` gave torch 2.14.0+cpu and
# torch.cuda.is_available() == False, silently, because cupy kept working.
# Cost: 0.71 TFLOP/s instead of 11.2 TFLOP/s (fp32).
# An exact +cu126 pin exists only in the PyTorch index, so there is no version
# contest left. Guarded by tests/test_requirements_torch_index_guard.py.
# macOS has no CUDA wheels and takes the plain build of the same version.
--extra-index-url https://download.pytorch.org/whl/cu126
torch==2.11.0+cu126 ; platform_system != "Darwin"
torchvision==0.26.0+cu126 ; platform_system != "Darwin"
torch==2.11.0 ; platform_system == "Darwin"
torchvision==0.26.0 ; platform_system == "Darwin"
scikit-image>=0.20
faiss-cpu

# torch.compile Inductor GPU codegen on Windows (community port; required for
# torch.compile mode="default" on CUDA in the GPU spherical-indexing hot path).
# Windows only; Linux users get Triton via the official PyTorch wheel and don't
# need this line.
triton-windows>=3.6,<3.7 ; sys_platform == "win32"

# CuPy + CUDA 12 runtime packages — alternative kernel-fusion path that bypasses
# Triton/Inductor (cupy.fuse multiply-and-maximum kernels in the GPU
# spherical indexer). cupy-cuda12x 14.x stopped bundling CUDA libs and requires
# the nvidia-* runtime wheels separately.
# macOS has no CUDA and no cupy-cuda12x / nvidia-*-cu12 wheels, so these are
# skipped there via an environment marker — otherwise `pip install -r
# requirements.txt` fails on Mac. On Linux/Windows they install as before; GPU
# code gates on torch.cuda.is_available() and falls back to CPU when no NVIDIA.
cupy-cuda12x>=14.0 ; platform_system != "Darwin"
nvidia-cuda-runtime-cu12 ; platform_system != "Darwin"
nvidia-cublas-cu12 ; platform_system != "Darwin"
nvidia-cufft-cu12 ; platform_system != "Darwin"
nvidia-curand-cu12 ; platform_system != "Darwin"
nvidia-cusolver-cu12 ; platform_system != "Darwin"
nvidia-cusparse-cu12 ; platform_system != "Darwin"
nvidia-cuda-nvrtc-cu12 ; platform_system != "Darwin"
